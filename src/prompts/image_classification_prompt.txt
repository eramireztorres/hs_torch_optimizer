    class FCNImageClassificationNN(nn.Module):
        def __init__(self, input_dim, hidden_dim, num_classes=10):
            super(FCNImageClassificationNN, self).__init__()

            self.fc2 = nn.Linear(hidden_dim, hidden_dim)

            return x
